# Ridder formulation

=LAMBDA(f, a, b, eps, [maxIter], [fa], [fb], [iter],
LET(
  maxIt, IF(ISOMITTED(maxIter), 200, maxIter),
  init, ISOMITTED(iter),

  faC, IF(init, f(a), fa),
  fbC, IF(init, f(b), fb),

  IF(
    init,
    IF(faC=0, a,
      IF(fbC=0, b,
        IF(faC*fbC>0,
          "#RIDDER! endpoints must bracket a root (opposite signs).",
          RIDDER(f, a, b, eps, maxIt, faC, fbC, 0)
        )
      )
    ),

    IF(
      OR((b-a)<=2*eps, iter>=maxIt),
      (a+b)/2,

      LET(
        m,  (a+b)/2,
        fm, f(m),

        s2, fm*fm - faC*fbC,

        x,  IF(s2<=0, m,
              m + (m-a) * SIGN(faC) * fm / SQRT(s2)
            ),
        fx, f(x),

        IF(
          fx=0, x,
          LET(
            cond1, fm*fx<0,
            cond2, faC*fx<0,

            newA,  IF(cond1, m, IF(cond2, a, x)),
            newB,  IF(cond1, x, IF(cond2, x, b)),
            newFa, IF(cond1, fm, IF(cond2, faC, fx)),
            newFb, IF(cond1, fx, IF(cond2, fx, fbC)),

            RIDDER(f, newA, newB, eps, maxIt, newFa, newFb, iter+1)
          )
        )
      )
    )
  )
))
